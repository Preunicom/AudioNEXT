\begin{frame}{Lösung / Implementierung - Audio - Takterzeugung [Hafeneder]}
    \begin{itemize}
        \item MCLK = 24,576 MHz 
        \item Aus MCLK wurden SCLK und LRCK durch Teilung erzeugt:
        \begin{itemize}
            \item \textbf{SCLK} = MCLK / 4 = 6,144 MHz \quad (SCLK/LRCK Ratio: 64)
            \item \textbf{LRCK} = MCLK / 256 = 96 kHz \quad (Double-Speed Mode)
        \end{itemize}
        \item Implementierung in \texttt{aud\_i2s\_clk\_gen.vhd}:
        \begin{itemize}
            \item SCLK: 1-Bit-Zähler togglet alle 2 MCLK-Takte \ra Periode = 4 MCLK
            \item LRCK: 7-Bit-Zähler togglet alle 128 MCLK-Takte \ra Periode = 256 MCLK
        \end{itemize}
        \item Takte nur aktiv wenn \texttt{i\_sen = '1'} (Sampling ist aktiviert)
    \end{itemize}
\end{frame}

\begin{frame}{Lösung / Implementierung - Audio - CDC \& Pufferung [Hafeneder]}
    \begin{itemize}
        \item Zwei Taktdomänen müssen überbrückt werden:
        \begin{itemize}
            \item \textbf{24,576 MHz} -- Audio (MCLK)
            \item \textbf{100 MHz} -- AXI Wrapper
        \end{itemize}
        \item \texttt{aud\_i2s\_sync}: 2-Flip-Flop Synchronisierer für \texttt{i\_data}
        \begin{itemize}
            \item Verhindert metastabile Zustände beim Wechsel der Taktdomäne
        \end{itemize}
        \item \texttt{aud\_channel\_buf}: AXI-Stream FIFO (Xilinx FIFO Generator 13.2)
        \begin{itemize}
            \item Schreibseite: 24,576 MHz (Audio)
            \item Leseseite: 100 MHz (AXI)
            \item Getrennte FIFOs für linken und rechten Kanal
        \end{itemize}
        \item Data Overrun Flag (DOL/DOR) wenn FIFO voll
    \end{itemize}
\end{frame}

\begin{frame}{Lösung / Implementierung - Audio AXI Hardware [Tuch]}
    \begin{itemize}
        \item AXI Lite Slave mit 10 Registern nach dem 5-Teile-Muster aus den Tutorials
        \item IPISR nutzt TOW (Toggle On Write): Schreiben einer 1 kippt das Statusbit per XOR zurück
        \item Interrupt-Ausgabe pro Kanal: GIE~$\wedge$~IPIER~$\wedge$~IPISR
        \item Lesen von ADATLR/ADATRR erzeugt einen Acknowledge-Puls, der den AXI-Stream Handshake mit dem CDC-FIFO auslöst
        \item Dabei wird \texttt{sla\_flag} für einen Takt maskiert, damit das IPISR-Bit nicht sofort erneut gesetzt wird
        \item Selftest prüft Register, Sampling Enable und beide Kanäle beim Programmstart
    \end{itemize}
    \vspace{0.3em}
    \centering
    \footnotesize
    \begin{tabular}{lll}
        \toprule
        \textbf{Offset} & \textbf{Register} & \textbf{Zugriff} \\
        \midrule
        0x00 & GSCR & Reserved \\
        0x04 & GIER (GIE) & R/W \\
        0x08 & IPIER (SLIE, SRIE) & R/W \\
        0x0C & IPISR (SLA, SRA) & R/W (TOW) \\
        0x10 & IDR & R (Konstante) \\
        0x14 & VERR & R (Konstante) \\
        0x18 & CTRL (SEN) & R/W \\
        0x1C & STATUS (DOL, DOR, SLA, SRA) & R \\
        0x20 & ADATLR (24\,Bit Audio) & R \\
        0x24 & ADATRR (24\,Bit Audio) & R \\
        \bottomrule
    \end{tabular}
\end{frame}
\begin{frame}{Lösung / Implementierung - Visualization AXI Hardware [Lonthoff]}
    \begin{columns}[T]
        \begin{column}{0.65\textwidth}
            \begin{itemize}
                \item AXI Lite Slave mit anderen Bits und einem Register (COLR) mehr als Audio AXI
                \item CTRL.WD wird von vis\_core automatisch gelöscht, sobald der Framebuffer die Übernahme bestätigt (\texttt{buf\_ready = '1'})
                \item Zeichenschreiben: ADDRR (X/Y) $\to$ VDATR (ASCII) $\to$ COLR (RGB) $\to$ CTRL.WD=1
                \item STATUS.FDP = 0 zeigt Ende des sichtbaren Frames $\to$ nächster Frame kann übertragen werden
                \item IPISR.FDP mit W1C Toggle: Schreibe 1 zum 0 setzen
            \end{itemize}

\begin{frame}{Lösung / Implementierung - VGA - Funktionsübersicht [Remy]}
    \begin{itemize}
        \item Bildaufteilung in 8x16 Pixel Blöcke für je ein Zeichen
        \item Schreiben neuer Werte in den Farb- \& Textframebuffer während des nicht sichtbaren Bereichs
        \item Textframebuffer: Verweis auf ASCII-Zeichen im ROM pro Block
        \item Farbframebuffer: 12 Bit Textfarbe pro Zeichenblock
        \item ROM: 8x16 Pixelmap pro Zeichen (1 = Text, 0 = Hintergrund)
        \item Pmod VGA zur Ausgabe des Signals
    \end{itemize}
\end{frame}
