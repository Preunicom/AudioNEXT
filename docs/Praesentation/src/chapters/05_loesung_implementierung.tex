\begin{frame}{Lösung / Implementierung - Audio - Takterzeugung [Hafeneder]}
    \begin{itemize}
        \item MCLK = 24,576 MHz 
        \item Aus MCLK wurden SCLK und LRCK durch Teilung erzeugt:
        \begin{itemize}
            \item \textbf{SCLK} = MCLK / 4 = 6,144 MHz \quad (SCLK/LRCK Ratio: 64)
            \item \textbf{LRCK} = MCLK / 256 = 96 kHz \quad (Double-Speed Mode)
        \end{itemize}
        \item Implementierung in \texttt{aud\_i2s\_clk\_gen.vhd}:
        \begin{itemize}
            \item SCLK: 1-Bit-Zähler togglet alle 2 MCLK-Takte \ra Periode = 4 MCLK
            \item LRCK: 7-Bit-Zähler togglet alle 128 MCLK-Takte \ra Periode = 256 MCLK
        \end{itemize}
        \item Takte nur aktiv wenn \texttt{i\_sen = '1'} (Sampling ist aktiviert)
    \end{itemize}
\end{frame}

\begin{frame}{Lösung / Implementierung - Audio - CDC \& Pufferung [Hafeneder]}
    \begin{itemize}
        \item Zwei Taktdomänen müssen überbrückt werden:
        \begin{itemize}
            \item \textbf{24,576 MHz} -- Audio (MCLK)
            \item \textbf{100 MHz} -- AXI Wrapper
        \end{itemize}
        \item \texttt{aud\_i2s\_sync}: 2-Flip-Flop Synchronisierer für \texttt{i\_data}
        \begin{itemize}
            \item Verhindert metastabile Zustände beim Wechsel der Taktdomäne
        \end{itemize}
        \item \texttt{aud\_channel\_buf}: AXI-Stream FIFO (Xilinx FIFO Generator 13.2)
        \begin{itemize}
            \item Schreibseite: 24,576 MHz (Audio)
            \item Leseseite: 100 MHz (AXI)
            \item Getrennte FIFOs für linken und rechten Kanal
        \end{itemize}
        \item Data Overrun Flag (DOL/DOR) wenn FIFO voll
    \end{itemize}
\end{frame}

\begin{frame}{Lösung / Implementierung - Audio AXI Hardware [Tuch]}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item AXI Lite Slave mit 10 Registern (5-Teile-Muster aus den Tutorials)
                \item IPISR mit W1C Toggle: Schreiben einer 1 kippt das Bit zurück
                \item Interrupt-Generierung: \texttt{GIE $\wedge$ IPIER $\wedge$ IPISR} pro Kanal
                \item Acknowledge-Puls beim Lesen von ADATLR/ADATRR löst AXI-Stream Handshake mit FIFO aus
                \item \texttt{sla\_flag} wird für einen Takt maskiert, damit IPISR nicht sofort erneut gesetzt wird
                \item Selftest prüft Register, Sampling Enable und beide Kanäle automatisch beim Start
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \centering
            \footnotesize
            \begin{tabular}{ll}
                \toprule
                \textbf{Offset} & \textbf{Register} \\
                \midrule
                0x00 & GSCR (Reserved) \\
                0x04 & GIER (GIE) \\
                0x08 & IPIER (SLIE, SRIE) \\
                0x0C & IPISR (SLA, SRA) \\
                0x10 & IDR (Konstante) \\
                0x14 & VERR (Konstante) \\
                0x18 & CTRL (SEN) \\
                0x1C & STATUS (DOL, DOR, SLA, SRA) \\
                0x20 & ADATLR (24\,Bit) \\
                0x24 & ADATRR (24\,Bit) \\
                \bottomrule
            \end{tabular}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Lösung / Implementierung - VGA - Funktionsübersicht [Remy]}
    \begin{itemize}
        \item Bildaufteilung in 8x16 Pixel Blöcke für je ein Zeichen
        \item Schreiben neuer Werte in den Farb- \& Textframebuffer während des nicht sichtbaren Bereichs
        \item Textframebuffer: Verweis auf ASCII-Zeichen im ROM pro Block
        \item Farbframebuffer: 12 Bit Textfarbe pro Zeichenblock
        \item ROM: 8x16 Pixelmap pro Zeichen (1 = Text, 0 = Hintergrund)
        \item Pmod VGA zur Ausgabe des Signals
    \end{itemize}
\end{frame}
