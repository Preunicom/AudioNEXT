\begin{frame}{Lösung / Implementierung - Audio AXI Hardware 2/2 [Tuch]}
    \begin{itemize}
        \item IPISR implementiert W1C als W1T: ermöglicht Selbsttest ohne echte Hardware-Events
        \item Lesen von ADATLR/ADATRR löst Acknowledge-Puls an den CDC-FIFO aus
        \item \texttt{sla\_flag} maskiert IPISR für einen Takt gegen erneutes Setzen
        \item Selftest prüft Register, Sampling Enable und beide Kanäle beim Start
    \end{itemize}
\end{frame}
\begin{frame}{Lösung / Implementierung - Visualization AXI Hardware [Lonthoff]}
    \begin{columns}[T]
        \begin{column}{0.65\textwidth}
            \begin{itemize}
                \item AXI Lite Slave mit anderen Bits und einem Register (COLR) mehr als Audio AXI
                \item CTRL.WD wird von vis\_core automatisch gelöscht, sobald der Framebuffer die Übernahme bestätigt (\texttt{buf\_ready = '1'})
                \item Zeichenschreiben: ADDRR (X/Y) $\to$ VDATR (ASCII) $\to$ COLR (RGB) $\to$ CTRL.WD=1
                \item STATUS.FDP = 0 zeigt Ende des sichtbaren Frames $\to$ nächster Frame kann übertragen werden
                \item IPISR.FDP mit W1C Toggle: Schreibe 1 zum 0 setzen
            \end{itemize}
        \end{column}
        \begin{column}{0.35\textwidth}
            \centering
            \footnotesize
            \begin{tabular}{ll}
                \toprule
                \textbf{Offset} & \textbf{Register} \\
                \midrule
                0x00 & GCSR (Reserved) \\
                0x04 & GIER (GIE) \\
                0x08 & IPIER (FIE) \\
                0x0C & IPISR (FDP) \\
                0x10 & IDR (Konstante) \\
                0x14 & VERR (Konstante) \\
                0x18 & CTRL (VEN, WD) \\
                0x1C & STATUS (FDP) \\
                0x20 & ADDRR (XA, YA) \\
                0x24 & VDATR (CHAR) \\
                0x28 & COLR (CR, CG, CB) \\
                \bottomrule
            \end{tabular}
        \end{column}
    \end{columns}
\end{frame}
\begin{frame}{Lösung / Implementierung - Visualization Core Software [Lonthoff]}
    \begin{columns}[T]
        \begin{column}{0.65\textwidth}
            \begin{itemize}
                \item Ringpuffer mit 80 Werten: ältester Wert links (Spalte~0), neuester rechts (Spalte~79)
                \item \texttt{VIS\_PollFDP} wartet auf STATUS.FDP vor Schreiben des nöchsten Frames
                \item \texttt{VIS\_WriteChar} schreibt ein farbiges (COLR) Zeichen (VDATR) an die aktuelle Position (ADDRR) und setzt CTRL.WD=1, dann \texttt{PollWD} bis zurückgesetzt

                \item Prozentanzeige: gleitender Mittelwert über die letzten vier 7.2-Festkommazahlen

\begin{frame}{Lösung / Implementierung - VGA - Funktionsübersicht [Remy]}
    \begin{itemize}
        \item Bildaufteilung in 8x16 Pixel Blöcke für je ein Zeichen
        \item Schreiben neuer Werte in den Farb- \& Textframebuffer während des nicht sichtbaren Bereichs
        \item Textframebuffer: Verweis auf ASCII-Zeichen im ROM pro Block
        \item Farbframebuffer: 12 Bit Textfarbe pro Zeichenblock
        \item ROM: 8x16 Pixelmap pro Zeichen (1 = Text, 0 = Hintergrund)
        \item Pmod VGA zur Ausgabe des Signals
    \end{itemize}
\end{frame}
