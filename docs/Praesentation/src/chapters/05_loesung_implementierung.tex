\begin{frame}{Lösung / Implementierung - VGA - Funktionsübersicht [Remy]}
    \begin{itemize}
        \item Bildaufteilung in 8x16 Pixel Blöcke für je ein Zeichen
        \item Schreiben neuer Werte in den Farb- \& Textframebuffer während des nicht sichtbaren Bereichs
        \item Textframebuffer: Verweis auf ASCII-Zeichen im ROM pro Block
        \item Farbframebuffer: 12 Bit Textfarbe pro Zeichenblock
        \item ROM: 8x16 Pixelmap pro Zeichen (1 = Text, 0 = Hintergrund)
        \item Pmod VGA zur Ausgabe des Signals
    \end{itemize}
\end{frame}
