\section{Diskussion}
\begin{frame}{Diskussion [Tuch]}
    \begin{itemize}
        \item \textbf{Bedeutung der Ergebnisse:}
        \begin{itemize}
            \item Vollständige Pipeline von der Audioquelle bis zur VGA-Anzeige funktionsfähig
            \item Eigene AXI IPs und Treiber zeigen das Zusammenspiel von HW und SW auf einem realen FPGA
            \item Selbsttests sichern die Kernfunktionen automatisch ab
        \end{itemize}
        \item \textbf{Ausblick:}
        \begin{itemize}
            \item Interrupt-gesteuertes Sampling statt Polling für geringere CPU-Last
            \item Frequenzanalyse (FFT) für ein Spektrogramm auf dem VGA-Ausgang
        \end{itemize}
        \item \textbf{Offene Punkte:}
        \begin{itemize}
            \item Auslastungsanalyse: Wie viel Puffer bleibt beim Schreiben in den CDC-FIFO, bevor Samples verloren gehen?
            \item Interrupt-Pfad ist in der Hardware vorbereitet (GIER, IPIER, IPISR), wird in Software aber noch nicht genutzt
        \end{itemize}
    \end{itemize}
\end{frame}
