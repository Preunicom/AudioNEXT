\section*{Notwendige Grundlagen}
\begin{frame}{Notwendige Grundlagen - I\textsuperscript{2}S Protokoll~\cite{digilent_pmod_i2s2, i2s_specification} [Hafeneder]}
    \begin{itemize}
        \item \textbf{I\textsuperscript{2}S}: Serielles digitales Audioprotokoll speziell zur Übertragung von Audiodaten zwischen integrierten Schaltkreisen wie etwa einem FPGA und dem CS5343 (A/D Wandler auf dem Pmod I2S2)
        \item verwendet drei Signalleitungen:
        \begin{itemize}
            \item \textbf{MCLK} (Master Clock): Haupttakt
            \item \textbf{LRCK} (Left-Right Clock / Word Select): Kanalauswahl (\texttt{'0'} = links, \texttt{'1'} = rechts)
            \item \textbf{SCLK} (Serial Clock / Bit Clock): Takt zur seriellen Datenübertragung
        \end{itemize}
        \item 32 Bit pro Kanal, MSB zuerst
        \item I\textsuperscript{2}S läuft im Master-Slave Prinzip \ra FPGA generiert entweder alle Taktsignale oder empfängt sie
    \end{itemize}
\end{frame}

\begin{frame}{Notwendige Grundlagen - RMS [Knoll]}
    \begin{itemize}
        \item RMS (Root Mean Square) ist der quadratische Mittelwert eines Signals
        \item Er beschreibt die mittlere Leistung und damit die wahrgenommene Lautstärke
        \item Werte quadrieren, über einen Block mitteln und die Wurzel ziehen
        \item Linker und rechter Kanal werden gemeinsam ausgewertet
    \end{itemize}
    \vspace{0.4em}
    \[
        x_{\mathrm{RMS}} = \sqrt{\frac{1}{2N} \sum_{i=1}^{N} \left( l_i^2 + r_i^2 \right)}
    \]
    \vspace{0.2em}
    \begin{itemize}
        \item Ergebnis als Prozent der Vollaussteuerung im 7.2 Festkommaformat
        \item Ganzzahlige Wurzel (isqrt) für die Berechnung ohne Gleitkomma
    \end{itemize}
\end{frame}
