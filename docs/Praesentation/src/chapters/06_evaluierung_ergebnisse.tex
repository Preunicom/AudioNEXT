\section{Evaluierung / Ergebnisse}
\begin{frame}{Evaluierung / Ergebnisse [Lonthoff]}
    \begin{itemize}
        \item System läuft stabil auf dem Arty A7-100T
        \item VGA-Ausgabe auf 3 verschiedenen Displays verifiziert (Beamer K101, K143, Monitor K143a)
        \item SW-Selftests automatisch beim Start: 11 Register, Formatierung (7 Grenzwerte), Balkenbreite (6 Fälle), Render (7 Werte), FDP-Protokoll
    \end{itemize}
