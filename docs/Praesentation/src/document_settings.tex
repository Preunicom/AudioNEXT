% ---------------------------------------------------------------------
%  Makros
% ---------------------------------------------------------------------
% Logo
\newcommand{\othlogo}[1][6mm]{\includegraphics[height=#1]{OTH_Logo_3line_pos_b.png}}

% Top border line
\newcommand{\othfullrule}{\rule{\paperwidth}{0.5pt}}
\newcommand{\othbleedrule}{\par\noindent\hspace*{-\othmargin}\othfullrule\par} % Ignore border

% Footer
\newcommand{\othfootfont}{\fontsize{4}{5}\selectfont}
\newcommand{\othfooterrow}{%
    \hbox to \paperwidth{%
        \hypersetup{linkcolor=gray}% Needed to have gray title as it is a hyperlink
        \othfootfont\normalfont\mdseries\color{gray}\strut%
        \ifnum\insertframenumber=1\else%
            \hspace*{\othmargin}\insertshorttitle\hfill%
            \insertshortdate\hfill%
            \insertframenumber\hspace*{\othmargin}%
        \fi%
    }%
}

% Bottom border line
\newcommand{\othfootband}{%
    \othfullrule\par%
    \vskip2.8mm%
    \othfooterrow%
    \vskip2.8mm%
}

% =====================================================================
%  Logo in top border as headline
% =====================================================================
\setbeamertemplate{headline}{%
    \vskip2mm%
    \hspace*{\othmargin}\othlogo\par%
    \vskip2mm%
    \othfullrule%
}

% Title
\setbeamertemplate{frametitle}{%
    \ifx\insertframetitle\empty\else%
        \vskip1.6mm%
        {\usebeamerfont{frametitle}\usebeamercolor[fg]{frametitle}\insertframetitle\par}%
        \vskip1mm%
    \fi%
}

% =====================================================================
%  Footer
% =====================================================================
\setbeamertemplate{footline}{%
    \othfootband%
}

% =====================================================================
%  Title page
% =====================================================================
% HAS TO BE USED AS NORMAL PAGE -> NO PLAIN PAGE (we use same border lines as normal pages)
\setbeamertemplate{title page}{%
    \vskip18mm%
    {\usebeamerfont{title}\usebeamercolor[fg]{title}\inserttitle\par}%
    \ifx\insertsubtitle\empty\else%
        \vskip2mm{\usebeamerfont{subtitle}\usebeamercolor[fg]{subtitle}\insertsubtitle\par}%
    \fi%
    \vskip12mm{\usebeamercolor[fg]{author}\small\insertauthor\par}%
    \vskip1.5mm{\usebeamercolor[fg]{institute}\footnotesize\insertinstitute\par}%
    \vskip1.5mm{\usebeamercolor[fg]{date}\footnotesize\insertdate\par}%
    \vfill
}
